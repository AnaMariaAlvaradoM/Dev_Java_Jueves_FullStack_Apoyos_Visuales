    @Override
    public UserDetails loadUserByUsername(String email) throws UsernameNotFoundException {
        Usuario usuario = usuarioRepository.findByEmail(email)
                .orElseThrow(() -> new UsernameNotFoundException("No existe el usuario " + email));

        return User.withUsername(usuario.getEmail())
                .password(usuario.getPassword())
                .authorities(autoridadesDe(usuario.getRol()))

export const jwtInterceptor: HttpInterceptorFn = (req, next) => {
  const authService = inject(AuthService);

  const esLogin = req.url.startsWith(`${API_URL}/auth/`);

import { Directive, TemplateRef, ViewContainerRef, effect, inject, input } from '@angular/core';
import { AuthService } from '../core/auth.service';
import { Permiso } from '../core/permisos';

// Dibuja su contenido si la persona tiene el permiso, o ALGUNO de la lista.
// Uso: *appSiPuede="'ESPACIOS_GESTIONAR'"
//      *appSiPuede="['RESERVAS_VER_TODAS', 'RESERVAS_VER_PROPIAS']"
@Directive({
  selector: '[appSiPuede]',
})
export class SiPuede {
  private authService = inject(AuthService);
  private plantilla = inject(TemplateRef);
  private contenedor = inject(ViewContainerRef);

  permiso = input.required<Permiso | Permiso[]>({ alias: 'appSiPuede' });

  private dibujado = false;

  constructor() {
    effect(() => {
      const valor = this.permiso();
      const lista = Array.isArray(valor) ? valor : [valor];
      const puede = lista.some(permiso => this.authService.puede(permiso));

      if (puede && !this.dibujado) {
        this.contenedor.createEmbeddedView(this.plantilla);
        this.dibujado = true;
      } else if (!puede && this.dibujado) {
        this.contenedor.clear();
        this.dibujado = false;
      }
    });
  }
}
